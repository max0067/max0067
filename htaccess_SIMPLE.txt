# Protection des fichiers sensibles
<FilesMatch "(\.py$|\.db$|\.sqlite$|requirements\.txt$|\.env$)">
    Order allow,deny
    Deny from all
</FilesMatch>

# Autoriser l'accès aux fichiers statiques
<FilesMatch "\.(css|js|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$">
    Order allow,deny
    Allow from all
</FilesMatch>

# Désactiver l'affichage des dossiers
